4 Okapi rebuild

The Okapi paper specifies the encoder and the sequence model. We chose the remaining settings.

Setting Okapi paper Our rebuild
Line embedding XLM-RoBERTa base, frozen; last four layers concatenated per token and averaged over the line (3,072 features) Same; lines cut at 128 subword tokens
Sequence model BiLSTM, one layer, 64 units; CRF output Same
Dropout and optimizer 0.25; RMSprop, learning rate 0.001 Same
Training emails Chipmunk’s English Gmane annotations annotations-final-train.jsonl (2,733 emails), split 90% training and 10% development with seed 13
Labels Chipmunk’s 15 zones The 15 zones plus <empty> for blank lines, assigned with the Chipmunk repository’s annotation_dict_to_lines, so scores compare with Chipmunk’s evaluate
Batch size and stopping Not reported 8 emails per batch; up to 50 epochs, stopping after 5 epochs without improvement on the development emails (training stopped after epoch 14 and kept epoch 9)

okapi.py embeds every line, trains the model, saves it as okapi-model.pt, and writes okapi-eval-gmane-validation.txt and okapi-eval-enron.txt.

cd ~/models/email-zoning
export HF_HOME=~/models/email-zoning/.cache/huggingface
source okapi-env/bin/activate
CUDA_VISIBLE_DEVICES=0 timeout 7200 python3 okapi.py 2>&1 | tee okapi-train.log
Noteokapi.py 
"""Okapi (Jardim, Rei and Almeida 2021), rebuilt from the paper's description.
Line labels follow annotation_dict_to_lines in webis-de/acl20-crawling-mailing-lists (src/util/mail_classification.py)."""
import json, random, sys
from pathlib import Path
import numpy as np, torch
from torch import nn
from torchcrf import CRF
from transformers import AutoTokenizer, AutoModel

SEED = 13
ROOT = Path(__file__).resolve().parent
ANN = ROOT / "acl20-crawling-mailing-lists" / "annotations"
REV = (ROOT / "xlmr-revision.txt").read_text().strip()
LABELS = ["paragraph", "closing", "inline_headers", "log_data", "mua_signature", "patch", "personal_signature",
          "quotation", "quotation_marker", "raw_code", "salutation", "section_heading", "tabular", "technical",
          "visual_separator", "<empty>"]
IDX = {l: i for i, l in enumerate(LABELS)}
EMPTY = IDX["<empty>"]

random.seed(SEED); np.random.seed(SEED); torch.manual_seed(SEED)
torch.backends.cudnn.deterministic, torch.backends.cudnn.benchmark = True, False
dev = torch.device("cuda")


def annotations(d):
    if "annotations" in d:
        return d["annotations"]
    return [{"start_offset": a[0], "end_offset": a[1], "label": a[2]} for a in d.get("labels") or []]


def line_labels(d):
    """Same logic as annotation_dict_to_lines in the Chipmunk repository."""
    lines = [l + "\n" for l in d["text"].split("\n")]
    anns = sorted(annotations(d), key=lambda a: a["start_offset"], reverse=True)
    offset, out = 0, []
    for l in lines:
        end = offset + len(l)
        if anns and offset > anns[-1]["end_offset"]:
            anns.pop()
        if not anns or not l.strip():
            out.append((l, "<empty>"))
        elif offset < anns[-1]["end_offset"] and end > anns[-1]["start_offset"]:
            out.append((l, anns[-1]["label"]))
        else:
            out.append((l, "<empty>"))
        offset = end
    return out


def load(name):
    mails = []
    for row in open(ANN / name):
        d = json.loads(row)
        if not annotations(d):
            continue
        ll = line_labels(d)
        if len(ll) > 5000:
            continue
        mails.append(([l.rstrip("\n") for l, _ in ll], [IDX[y] for _, y in ll]))
    return mails


tok = AutoTokenizer.from_pretrained("FacebookAI/xlm-roberta-base", revision=REV)
enc = AutoModel.from_pretrained("FacebookAI/xlm-roberta-base", revision=REV).to(dev).eval()


@torch.no_grad()
def embed(lines, bs=256):
    out = []
    for i in range(0, len(lines), bs):
        b = tok(lines[i:i + bs], padding=True, truncation=True, max_length=128, return_tensors="pt").to(dev)
        hs = enc(**b, output_hidden_states=True).hidden_states
        x = torch.cat(hs[-4:], dim=-1)
        m = b["attention_mask"].unsqueeze(-1).float()
        out.append(((x * m).sum(1) / m.sum(1)).half().cpu())
    return torch.cat(out)


def embed_set(name):
    cache = ROOT / "okapi-cache" / (name + ".pt")
    if cache.exists():
        return torch.load(cache)
    mails = load(name)
    data = [(embed(lines), torch.tensor(y)) for lines, y in mails]
    cache.parent.mkdir(exist_ok=True)
    torch.save(data, cache)
    return data


class Okapi(nn.Module):
    def __init__(self):
        super().__init__()
        self.lstm = nn.LSTM(3072, 64, num_layers=1, batch_first=True, bidirectional=True)
        self.drop = nn.Dropout(0.25)
        self.out = nn.Linear(128, len(LABELS))
        self.crf = CRF(len(LABELS), batch_first=True)

    def emissions(self, x):
        return self.out(self.drop(self.lstm(x)[0]))


def batch(items):
    n = max(len(y) for _, y in items)
    x = torch.zeros(len(items), n, 3072)
    y = torch.zeros(len(items), n, dtype=torch.long)
    m = torch.zeros(len(items), n, dtype=torch.bool)
    for i, (xe, ye) in enumerate(items):
        x[i, :len(ye)], y[i, :len(ye)], m[i, :len(ye)] = xe.float(), ye, True
    return x.to(dev), y.to(dev), m.to(dev)


@torch.no_grad()
def predict(model, data):
    model.eval()
    gold, pred = [], []
    for i in range(0, len(data), 8):
        x, y, m = batch(data[i:i + 8])
        for p, yy, mm in zip(model.crf.decode(model.emissions(x), m), y, m):
            pred.extend(p); gold.extend(yy[mm].tolist())
    return np.array(gold), np.array(pred)


def report(name, gold, pred):
    cm = np.zeros((len(LABELS), len(LABELS)))
    for g, p in zip(gold, pred):
        cm[g, p] += 1
    ne = gold != EMPTY
    lines = [f"{name}: {len(gold)} lines",
             f"accuracy, all lines (as Chipmunk's evaluate): {(gold == pred).mean():.4f}",
             f"accuracy, non-blank lines: {(gold[ne] == pred[ne]).mean():.4f}",
             "", f"{'zone':<20}{'lines':>8}{'recall':>8}  main confusion"]
    for i, l in enumerate(LABELS):
        n = cm[i].sum()
        if n == 0:
            continue
        row = cm[i] / n
        off = [(row[j], LABELS[j]) for j in range(len(LABELS)) if j != i and row[j] > 0]
        top = max(off) if off else (0, "")
        lines.append(f"{l:<20}{int(n):>8}{row[i]:>8.3f}  {top[1]} {top[0]:.3f}" if top[1] else f"{l:<20}{int(n):>8}{row[i]:>8.3f}")
    text = "\n".join(lines)
    print(text + "\n")
    (ROOT / f"okapi-eval-{name}.txt").write_text(text + "\n")


train_all = embed_set("annotations-final-train.jsonl")
order = list(range(len(train_all)))
random.Random(SEED).shuffle(order)
cut = len(order) // 10
devset = [train_all[i] for i in order[:cut]]
train = [train_all[i] for i in order[cut:]]
print(f"train {len(train)} emails | dev {len(devset)} emails")

model = Okapi().to(dev)
opt = torch.optim.RMSprop(model.parameters(), lr=0.001)
best, best_state, wait = -1, None, 0
for epoch in range(1, 51):
    model.train()
    random.Random(SEED + epoch).shuffle(train)
    for i in range(0, len(train), 8):
        x, y, m = batch(train[i:i + 8])
        loss = -model.crf(model.emissions(x), y, m, reduction="mean")
        opt.zero_grad(); loss.backward(); opt.step()
    g, p = predict(model, devset)
    acc = (g == p).mean()
    print(f"epoch {epoch:>2} | dev accuracy {acc:.4f}", flush=True)
    if acc > best:
        best, best_state, wait = acc, {k: v.detach().clone() for k, v in model.state_dict().items()}, 0
    else:
        wait += 1
        if wait >= 5:
            break
model.load_state_dict(best_state)
torch.save(best_state, ROOT / "okapi-model.pt")

report("gmane-validation", *predict(model, embed_set("annotations-final-validation.jsonl")))
report("enron", *predict(model, embed_set("enron-annotated.jsonl")))

5 Recall by zone on Enron

6 Five emails

five-emails.txt holds the first five emails of the 30-email sample. This block splits it into one body per email in emails/, named by msgid2, and saves the header fields to emails/headers.json. Each body is everything below the ─── line, unchanged.

cd ~/models/email-zoning
source okapi-env/bin/activate
timeout 60 python3 - <<'EOF'
import json, re
from pathlib import Path
raw = Path("five-emails.txt").read_text()
blocks = re.split(r"^(?=Email \d+ of \d+$)", raw, flags=re.M)
out, headers = Path("emails"), {}
out.mkdir(exist_ok=True)
for b in (b for b in blocks if b.strip()):
    head, body = re.split(r"^─{20,}\n", b, maxsplit=1, flags=re.M)
    fields = dict(l.split(": ", 1) for l in head.strip().split("\n")[1:])
    fields["Email"] = head.split("\n", 1)[0]
    msgid = fields["msgid2"]
    (out / f"{msgid}.txt").write_text(body.rstrip("\n") + "\n")
    headers[msgid] = fields
    print(msgid, fields["Email"], "|", len(body.strip().split("\n")), "lines")
(out / "headers.json").write_text(json.dumps(headers, indent=2, ensure_ascii=False))
EOF
deactivate

Chipmunk labels each line with the repository’s own functions and writes zones-chipmunk.jsonl. chipmunk_raw holds the model’s top zone for each line; chipmunk holds the same after the repository’s _post_process_labels corrections.

cd ~/models/email-zoning
source chipmunk-env/bin/activate
CUDA_VISIBLE_DEVICES="" TF_CPP_MIN_LOG_LEVEL=3 PYTHONPATH=acl20-crawling-mailing-lists/src PYTHONWARNINGS=ignore timeout 1800 python3 - <<'EOF'
import json
from pathlib import Path
import numpy as np
from tensorflow.keras import models
from util.mail_classification import load_fasttext_model, MailLinesSequence
from parsing.message_segmenter import CONTEXT_SHAPE, INF_BATCH_SIZE, _post_process_labels

load_fasttext_model("weights/fasttext-model.bin")
model = models.load_model("weights/segmenter_hinge.epoch-11.val_loss-0.093.h5")
headers = json.loads(Path("emails/headers.json").read_text())
rows = []
for msgid, h in headers.items():
    text = Path(f"emails/{msgid}.txt").read_text().rstrip("\n")
    seq = MailLinesSequence(text, CONTEXT_SHAPE, labeled=False, input_is_raw_text=True, batch_size=INF_BATCH_SIZE)
    soft = model.predict(seq)
    raw = [MailLinesSequence.LABEL_MAP_INVERSE[int(np.argmax(s))] for s in soft]
    post = [label for _, label in _post_process_labels(seq, soft.copy())]
    print(f"\n{h['Email']} (msgid2 {msgid})")
    for i, (line, r, p) in enumerate(zip(text.split("\n"), raw, post), 1):
        rows.append({"msgid2": msgid, "line": i, "text": line, "chipmunk_raw": r, "chipmunk": p})
        print(f"{i:>3}  {r:<19}{p:<19}{line[:70]}")
Path("zones-chipmunk.jsonl").write_text("".join(json.dumps(r, ensure_ascii=False) + "\n" for r in rows))
EOF

Okapi loads okapi-model.pt, labels each line, and merges its labels with Chipmunk’s into zones.jsonl. Each embedding passes through half precision before the BiLSTM, as in training.

cd ~/models/email-zoning
source okapi-env/bin/activate
export HF_HOME=~/models/email-zoning/.cache/huggingface
CUDA_VISIBLE_DEVICES=0 timeout 900 python3 - <<'EOF'
import json
from pathlib import Path
import torch
from torch import nn
from torchcrf import CRF
from transformers import AutoTokenizer, AutoModel

LABELS = ["paragraph", "closing", "inline_headers", "log_data", "mua_signature", "patch", "personal_signature",
          "quotation", "quotation_marker", "raw_code", "salutation", "section_heading", "tabular", "technical",
          "visual_separator", "<empty>"]
REV = Path("xlmr-revision.txt").read_text().strip()
dev = torch.device("cuda")
tok = AutoTokenizer.from_pretrained("FacebookAI/xlm-roberta-base", revision=REV)
enc = AutoModel.from_pretrained("FacebookAI/xlm-roberta-base", revision=REV).to(dev).eval()


class Okapi(nn.Module):
    """Same architecture as okapi.py."""
    def __init__(self):
        super().__init__()
        self.lstm = nn.LSTM(3072, 64, num_layers=1, batch_first=True, bidirectional=True)
        self.drop = nn.Dropout(0.25)
        self.out = nn.Linear(128, len(LABELS))
        self.crf = CRF(len(LABELS), batch_first=True)

    def emissions(self, x):
        return self.out(self.drop(self.lstm(x)[0]))


@torch.no_grad()
def embed(lines):
    b = tok(lines, padding=True, truncation=True, max_length=128, return_tensors="pt").to(dev)
    x = torch.cat(enc(**b, output_hidden_states=True).hidden_states[-4:], dim=-1)
    m = b["attention_mask"].unsqueeze(-1).float()
    return ((x * m).sum(1) / m.sum(1)).half().float()


model = Okapi().to(dev)
model.load_state_dict(torch.load("okapi-model.pt"))
model.eval()

rows = [json.loads(l) for l in open("zones-chipmunk.jsonl")]
headers = json.loads(Path("emails/headers.json").read_text())
for msgid, h in headers.items():
    mine = [r for r in rows if r["msgid2"] == msgid]
    x = embed([r["text"] for r in mine]).unsqueeze(0)
    pred = model.crf.decode(model.emissions(x), torch.ones(x.shape[:2], dtype=torch.bool, device=dev))[0]
    print(f"\n{h['Email']} (msgid2 {msgid})")
    print(f"{'':>3}  {'chipmunk_raw':<19}{'chipmunk':<19}{'okapi':<19}text")
    for r, p in zip(mine, pred):
        r["okapi"] = LABELS[p]
        print(f"{r['line']:>3}  {r['chipmunk_raw']:<19}{r['chipmunk']:<19}{r['okapi']:<19}{r['text'][:60]}")
Path("zones.jsonl").write_text("".join(json.dumps(r, ensure_ascii=False) + "\n" for r in rows))
EOF
